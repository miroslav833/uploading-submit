Verification notes: quasi-Fermi-level splitting from an absolute photoluminescence image (ch92)

1. What is here

The figure is a simulated absolute photoluminescence image of a perovskite film under a Gaussian beam with four printed lines of uniform emission, drawn by image_source.py from the design in verification/layout.py (the film and line levels, all at band centres). Section 2 integrates the generalized Planck law numerically and checks the closed form. Section 3 is the key in closed form with the wrong routes. Sections 4 and 5 read the band of every line back from the pixels of the image file at full size and at 70 percent. The key is known by construction (line 3 was drawn at 10^14.68), by the closed form and by the read-back.

Environment: Python 3.12.3; numpy 2.5.5, SciPy 1.18.1, RDKit 2026.09.2, PySCF 2.15.0. Every program below is complete as printed and was run exactly as shown.

2. Generalized Planck law: numerical integral and closed form
=============================================================

Command (run inside verification/): python3 model.py 14.63 14.68 14.73

----- model.py -----
"""Quasi-Fermi-level splitting from an absolute PL photon flux (numpy, scipy).

Generalized Planck law (Wurfel 1982) for the spectrally integrated photon flux leaving the front surface
into the hemisphere, Boltzmann limit E - dmu >> kT:
    Phi_PL = exp(dmu/kT) * Phi_0,   Phi_0 = int a(E) (2 pi / (h^3 c^2)) E^2 exp(-E/kT) dE
a(E) = a0 for E >= Eg and a0 exp((E - Eg)/Eu) below (Urbach tail).
"""
import math, sys, json
import numpy as np
from scipy.integrate import quad

h = 6.62607015e-34; c = 2.99792458e8; kB = 1.380649e-23; q = 1.602176634e-19
P = dict(T=296.0, a0=0.86, Eg=1.614, Eu=0.0142)


def phi0(T, a0, Eg, Eu, tail=True, pref=2 * math.pi, bose=False):
    kT = kB * T / q                     # eV
    C = pref / (h ** 3 * c ** 2) * q ** 3   # photons m^-2 s^-1 per eV, with E in eV
    def bb(E):
        return C * E * E / (math.expm1(E / kT) if bose else math.exp(E / kT))
    above = quad(lambda E: a0 * bb(E), Eg, Eg + 60 * kT, limit=200)[0]
    below = quad(lambda E: a0 * math.exp((E - Eg) / Eu) * bb(E), Eg - 80 * Eu, Eg, limit=200)[0] if tail else 0.0
    return (above + below) * 1e-4, above * 1e-4, below * 1e-4   # photons cm^-2 s^-1


def closed(T, a0, Eg, Eu):
    """Same integral in closed form (Boltzmann factor, tail integrated to -infinity)."""
    kT = kB * T / q
    C = 2 * math.pi / (h ** 3 * c ** 2) * q ** 3 * 1e-4
    up = a0 * C * math.exp(-Eg / kT) * kT * (Eg ** 2 + 2 * Eg * kT + 2 * kT ** 2)
    g = 1 / Eu - 1 / kT                 # tail: int E^2 exp(g (E - Eg)) dE from -inf to Eg times exp(-Eg/kT)
    dn = a0 * C * math.exp(-Eg / kT) * (Eg ** 2 / g - 2 * Eg / g ** 2 + 2 / g ** 3)
    return up + dn, up, dn


def dmu(logphi, **kw):
    p = dict(P); p.update(kw)
    tot = phi0(p["T"], p["a0"], p["Eg"], p["Eu"])[0]
    kT = kB * p["T"] / q
    return kT * (logphi * math.log(10) - math.log(tot))


if __name__ == "__main__":
    T, a0, Eg, Eu = P["T"], P["a0"], P["Eg"], P["Eu"]
    kT = kB * T / q
    tot, up, dn = phi0(T, a0, Eg, Eu)
    ct, cu, cd = closed(T, a0, Eg, Eu)
    print("kT = %.6f eV" % kT)
    print("Phi_0 numerical: total %.5e, above Eg %.5e, Urbach tail %.5e photons cm^-2 s^-1" % (tot, up, dn))
    print("Phi_0 closed form: total %.5e, above Eg %.5e, tail %.5e" % (ct, cu, cd))
    print("tail / above = %.4f" % (dn / up))
    print("Bose-Einstein instead of Boltzmann changes Phi_0 by %.2e (relative)" % (phi0(T, a0, Eg, Eu, bose=True)[0] / tot - 1))
    for lg in [float(x) for x in sys.argv[1:]] or [14.3]:
        print("log10 Phi_PL = %.2f -> dmu = %.5f eV" % (lg, dmu(lg)))
----- end of model.py -----

def phi0(T=T, a0=a0, Eg=Eg, Eu=Eu, pref=1.0, tail=True):
    kT = kB * T / q
    up = a0 * C * math.exp(-Eg / kT) * kT * (Eg ** 2 + 2 * Eg * kT + 2 * kT ** 2)
    g = 1 / Eu - 1 / kT
    dn = a0 * C * math.exp(-Eg / kT) * (Eg ** 2 / g - 2 * Eg / g ** 2 + 2 / g ** 3) if tail else 0.0
    return pref * (up + dn), pref * up, pref * dn

"""Reads the band of every labelled spot back from image.png (Pillow, standard library otherwise).
The palette is sampled at the middle of each block of the bar on the right; each printed line is sampled
along its centre line. The layout gives only where to sample; the band comes from the pixels."""
import os, sys, statistics
from PIL import Image
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import layout as L

path = sys.argv[1] if len(sys.argv) > 1 else os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), "image.png")
im = Image.open(path).convert("L")
if len(sys.argv) > 2:                            # optional scale factor: read a reduced copy
    f = float(sys.argv[2])
    im = im.resize((int(im.size[0] * f), int(im.size[1] * f)), Image.LANCZOS)
    print("reading a copy scaled to %.0f %%, %d x %d px" % (100 * f, im.size[0], im.size[1]))
W, H = im.size
s = W / 1200.0                                   # works on scaled copies too
px = im.load()
X0, Y0, MW, MH = 50 * s, 70 * s, L.W_MM * L.PX_MM * s, L.H_MM * L.PX_MM * s
NB = int(round((L.HI - L.LO) / L.BAND))
BX, BW = X0 + MW + 60 * s, 42 * s
bh = MH / NB
pal = []
for k in range(NB):
    yc = Y0 + MH - (k + 0.5) * bh
    vals = [px[int(BX + BW * f), int(yc + dy)] for f in (0.3, 0.5, 0.7) for dy in (-3, 0, 3)]
    pal.append(statistics.median(vals))
print("palette grays, lowest band first:", pal)
import math
for lab, x0, y0, ln, ang, v in L.SPOTS:
    # sample the central pixel row of each printed line, from 10 % to 90 % of its length
    ux, uy = math.cos(math.radians(ang)), -math.sin(math.radians(ang))
    vals = []
    for i in range(200):
        t = (-0.4 + 0.8 * i / 199) * ln
        xi, yi = X0 + (x0 + t * ux) * L.PX_MM * s, Y0 + (y0 + t * uy) * L.PX_MM * s
        vals.append(px[int(xi), int(yi)])
    g = statistics.median(vals)
    k = min(range(NB), key=lambda kk: abs(pal[kk] - g))
    print("line %s: median gray %.0f along the line (10th-90th percentile %d-%d), nearest palette band %d, centre %.2f (design %.2f) %s" %
          (lab, g, sorted(vals)[20], sorted(vals)[179], k, L.band_centre(k), v, "OK" if abs(L.band_centre(k) - v) < 1e-6 else "MISMATCH"))

6. layout.py
============

"""Design of the PL map: film field and treated spots (standard library). Imported by image_source.py,
key.py and readback_png.py. All values are log10 of the photon flux in photons cm^-2 s^-1."""
import math
W_MM, H_MM = 12.0, 9.0          # field of view
PX_MM = 75.0                    # pixels per mm in the map
CX, CY = 5.55, 4.70             # beam centre, mm from the left and top
PEAK, CURV = 15.55, 0.075       # film: log10 Phi = PEAK - CURV * r^2 (Gaussian excitation, PL ~ G^1.5)
BAND = 0.05
LO, HI = 13.005, 16.005         # palette range, 60 bands; band centres 13.03, 13.08, ...
SPOTS = [  # label, x mm, y mm of the centre, length mm, angle deg (from +x, counter-clockwise on screen), value
    ("1", 4.10, 2.60, 2.4, 20.0, 15.88),
    ("2", 7.40, 4.30, 2.4, 70.0, 15.18),
    ("3", 9.45, 7.15, 2.4, 128.0, 14.68),
    ("4", 2.20, 6.30, 2.4, 95.0, 13.88),
]
LINE_W = 0.067                  # printed line width, mm (5 px)
KEY_SPOT = "3"


def film(x, y):
    return PEAK - CURV * ((x - CX) ** 2 + (y - CY) ** 2)


def band_index(v):
    return int(math.floor((v - LO) / BAND + 1e-9))


def band_centre(k):
    return LO + (k + 0.5) * BAND


def on_line(xm, ym, sp):
    """Mask of points on the printed line sp (works on numpy arrays or floats)."""
    lab, x0, y0, ln, ang, v = sp
    ux, uy = math.cos(math.radians(ang)), -math.sin(math.radians(ang))
    t = (xm - x0) * ux + (ym - y0) * uy
    n = -(xm - x0) * uy + (ym - y0) * ux
    return (abs(t) <= ln / 2) & (abs(n) <= LINE_W / 2)
