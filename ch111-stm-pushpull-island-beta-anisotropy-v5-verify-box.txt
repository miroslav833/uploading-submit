def sym(p, q, r): return [p[0]*q[0]*r[0], (p[0]*q[0]*r[1] + p[0]*q[1]*r[0] + p[1]*q[0]*r[0])/3, (p[0]*q[1]*r[1] + p[1]*q[0]*r[1] + p[1]*q[1]*r[0])/3, p[1]*q[1]*r[1]]
n = {0: 5, 60: 6, 120: 4, 180: 7, 240: 7, 300: 11}
con = [(300,180,1), (60,0,-1), (0,300,-1), (0,180,1), (240,180,-1), (180,180,-1), (60,180,1), (240,300,-1), (300,300,-1), (300,300,-1), (0,0,-1), (0,300,-1), (120,60,-1), (60,300,1), (300,120,1), (120,60,-1), (60,60,-1), (240,300,-1), (240,120,1), (60,240,1)]
