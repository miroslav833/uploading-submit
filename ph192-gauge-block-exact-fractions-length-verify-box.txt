import math, cmath
lam = [632.9908, 543.5164, 514.6734, 488.1225]
na = [1.00026987, 1.00027127, 1.00027190, 1.00027258]
nk = [2.86+3.37j, 2.69+3.12j, 2.62+3.04j, 2.55+2.96j]
f = [0.214, 0.483, 0.591, 0.455]
def F(d, i):
    a = cmath.phase((1 - nk[i]) / (1 + nk[i])) % (2 * math.pi) - math.pi
    return 2 * na[i] * ((75e6 + d) * (1 + 11.5e-6 * 0.36) - a * lam[i] / (4 * math.pi)) / lam[i]
mis = lambda d: max(min((F(d, i) - f[i]) % 1, 1 - (F(d, i) - f[i]) % 1) for i in range(4))
d = min((0.5 * x for x in range(-3200, 6801)), key=mis)
print("deviation %.1f nm, misfit %.3f" % (d, mis(d)))
