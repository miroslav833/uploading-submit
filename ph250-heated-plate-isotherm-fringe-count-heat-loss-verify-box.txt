import math
def rk(Pr, a, b, E=10.0, n=4000):
    h, s = E / n, [0.0, 0.0, a, 1.0, b]
    F = lambda s: [s[1], s[2], -3*s[0]*s[2] + 2*s[1]**2 - s[3], s[4], -3*Pr*s[0]*s[4]]
    for _ in range(n):
        k1 = F(s); k2 = F([u + h/2*v for u, v in zip(s, k1)])
        k3 = F([u + h/2*v for u, v in zip(s, k2)]); k4 = F([u + h*v for u, v in zip(s, k3)])
        s = [u + h/6*(p + 2*q + 2*r + w) for u, p, q, r, w in zip(s, k1, k2, k3, k4)]
    return s[1], s[3]
def shoot(Pr, a=0.67, b=-0.51):
    for _ in range(30):
        f0 = rk(Pr, a, b); d = 1e-6
        fa = rk(Pr, a + d, b); fb = rk(Pr, a, b + d)
        J = [[(fa[0]-f0[0])/d, (fb[0]-f0[0])/d], [(fa[1]-f0[1])/d, (fb[1]-f0[1])/d]]
        det = J[0][0]*J[1][1] - J[0][1]*J[1][0]
        a -= ( J[1][1]*f0[0] - J[0][1]*f0[1]) / det
        b -= (-J[1][0]*f0[0] + J[0][0]*f0[1]) / det
    return b
