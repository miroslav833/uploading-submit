Check program (Python standard library, prints vf 0.31161, j0,43 134.3040, w0 7.4354 um, W 0.5098 uW):
import math
J = lambda n, x, s: sum(s**k * (x/2)**(2*k+n) / (math.factorial(k) * math.factorial(k+n)) for k in range(40))
L = 3.19622
c = J(0, L, -1) / J(0, L, 1)
vf = 4 * J(1, L, -1) / (L * (1 - c))
b = (43 - 0.25) * math.pi
z43 = b + 1/(8*b) - 31/(384*b**3)
K = 2*math.pi/0.6328 * (1 + math.cos(math.radians(35.0)))
w0 = z43 / K * 1e-6
om = 2*math.pi*813.0
Q = om * vf * math.pi * 0.015**2 * w0
W = 1.204 * 343.2 * (om/343.2)**2 * Q**2 / (4*math.pi)
print("vf %.5f  j0,43 %.4f  w0 %.4f um  W %.4f uW" % (vf, z43, w0*1e6, W*1e6))
