rows = '''...'''
g = [l.split()[2:] for l in rows.strip().splitlines()]
A = {(i, j) for i in range(9) for j in range(9) if g[i][2 * j] == "Al"}
n = [0] * 5
for i in range(9):
    for j in range(9):
        if (i, j) not in A:
            n[sum(((i + a) % 9, (j + b) % 9) in A for a, b in ((1, 0), (-1, 0), (0, 1), (0, -1)))] += 1
print(n, round(100 * n[2] / sum(n), 1))
