Verification notes: coupling-induced change of the nonlinear anisotropy of forty push-pull molecules (ch111, version 4)

1. What is here

The key from the closed first-order form with the values of the wrong routes and of misread molecules, the same key from exact diagonalisation of every aggregate (for two values of the dipole change), and a readback of directions and contacts from the PNG. The layout is embedded at the end.

Environment: Python 3.12.3; numpy 2.5.5, SciPy 1.18.1. Every program below is complete as printed and was run exactly as shown.

2. Key from the closed form, wrong routes, misreads
===================================================

Command (run inside verification/): python3 key.py

----- key.py -----
"""Key and wrong routes for ch111 version 4 (standard library only), from verification/layout.json.
Units: beta0 = 1, x = J/eps. Zeroth order B0 = sum_a u_a u_a u_a. A contact a-b with sign s adds, to first order
and exactly, -6 s x Sym[u_a u_b u_b + u_a u_a u_b] (first-order energy s (J/mu^2) p_a(F) p_b(F) with
p = alpha f + beta0 f^2 / 2, alpha = 2 mu^2 / eps). s is the sign of u_a.u_b - 3 (u_a.r)(u_b.r), r along the line of
centres. rho^2 = (5/3) |B|^2 / |v|^2 - 1, kappa = d rho / dx at x = 0."""
import json
import math
import os
import sys

p = sys.argv[1] if len(sys.argv) > 1 else os.path.join(os.path.dirname(os.path.abspath(__file__)), "layout.json")
L = json.load(open(p))
ang = [m["u_deg_lattice"] for m in L["molecules"]]
cen = [m["centre_px"] for m in L["molecules"]]
rot = L["rot_deg"]


def uv(d):
    return (math.cos(math.radians(d)), math.sin(math.radians(d)))


def sym(p_, q, r):
    return [p_[0] * q[0] * r[0],
            (p_[0] * q[0] * r[1] + p_[0] * q[1] * r[0] + p_[1] * q[0] * r[0]) / 3,
            (p_[0] * q[1] * r[1] + p_[1] * q[0] * r[1] + p_[1] * q[1] * r[0]) / 3,
            p_[1] * q[1] * r[1]]


def sign(a, b, ang_):
    ua, ub = uv(ang_[a] + rot), uv(ang_[b] + rot)
    rx, ry = cen[b][0] - cen[a][0], -(cen[b][1] - cen[a][1])           # image y runs down
    n = math.hypot(rx, ry)
    rx, ry = rx / n, ry / n
    return 1 if ua[0] * ub[0] + ua[1] * ub[1] - 3 * (ua[0] * rx + ua[1] * ry) * (ub[0] * rx + ub[1] * ry) > 0 else -1


def tensors(ang_, signs=None, factor=6.0):
    U = [uv(d) for d in ang_]
    B0 = [0.0] * 4
    for u in U:
        B0 = [x + y for x, y in zip(B0, sym(u, u, u))]
    B1 = [0.0] * 4
    for i, (a, b) in enumerate(L["contacts"]):
        s = signs[i] if signs else sign(a, b, ang_)
        t = [x + y for x, y in zip(sym(U[a], U[b], U[b]), sym(U[a], U[a], U[b]))]
        B1 = [x - factor * s * y for x, y in zip(B1, t)]
    return B0, B1


W4 = (1, 3, 3, 1)


def kappa(B0, B1, show=False):
    v0 = (B0[0] + B0[2], B0[1] + B0[3])
    v1 = (B1[0] + B1[2], B1[1] + B1[3])
    n0 = sum(w * x * x for w, x in zip(W4, B0))
    n1 = 2 * sum(w * x * y for w, x, y in zip(W4, B0, B1))
    q0 = v0[0] ** 2 + v0[1] ** 2
    q1 = 2 * (v0[0] * v1[0] + v0[1] * v1[1])
    rho0 = math.sqrt(5 * n0 / (3 * q0) - 1)
    k = (5 / 3) * (n1 / q0 - n0 * q1 / q0 ** 2) / (2 * rho0)
    if show:
        print("B0 (xxx, xxy, xyy, yyy) = (%.4f, %.4f, %.4f, %.4f)" % tuple(B0))
        print("B1 (xxx, xxy, xyy, yyy) = (%.4f, %.4f, %.4f, %.4f)" % tuple(B1))
        print("|B0|^2 = %.4f, d|B|^2/dx = %.4f, |v0|^2 = %.4f, d|v|^2/dx = %.4f" % (n0, n1, q0, q1))
        print("rho0 = %.4f, d(rho^2)/dx = %.4f" % (rho0, 2 * rho0 * k))
    return rho0, k


counts = [ang.count(60 * k) for k in range(6)]
print("molecules", len(ang), "by direction 0..300:", counts, "contacts", len(L["contacts"]))
sg = [sign(a, b, ang) for a, b in L["contacts"]]
print("contacts with s = -1 (head to tail):", sg.count(-1), " with s = +1 (like ends meet):", sg.count(1))
B0, B1 = tensors(ang)
rho0, key = kappa(B0, B1, show=True)
print("KEY kappa = %.4f" % key)
print("Heitler-London exciton model (one third of the first-order change): %.4f" % kappa(*tensors(ang, factor=2.0))[1])
print("local-field factor squared instead of cubed (factor 4): %.4f" % kappa(*tensors(ang, factor=4.0))[1])
print("all couplings negative: %.4f" % kappa(*tensors(ang, signs=[-1] * len(sg)))[1])
print("all couplings positive: %.4f" % kappa(*tensors(ang, signs=[1] * len(sg)))[1])
print("every sign reversed: %.4f" % kappa(*tensors(ang, signs=[-s for s in sg]))[1])
ks = "%.3g" % key
one = []
for i in range(len(ang)):
    a2 = ang[:]
    a2[i] = (a2[i] + 180) % 360
    one.append("%.3g" % kappa(*tensors(a2))[1])
print("one molecule read end for end: %d cases, values from %s to %s, key among them: %s" % (len(one), min(one, key=float), max(one, key=float), ks in one))
near = sorted(one, key=lambda v: abs(float(v) - key))[:4]
print("  nearest to the key:", near)
two = 0
n = len(ang)
for i in range(n):
    for j in range(i + 1, n):
        a2 = ang[:]
        a2[i] = (a2[i] + 180) % 360
        a2[j] = (a2[j] + 180) % 360
        if "%.3g" % kappa(*tensors(a2))[1] == ks:
            two += 1
print("pairs of molecules read end for end that give the key again: %d of %d" % (two, n * (n - 1) // 2))

3. Key from exact diagonalisation
=================================

Command (run inside verification/): python3 exact.py

----- exact.py -----
"""Independent route: exact diagonalisation. For every aggregate the full Hamiltonian of its two-level molecules is
built (units mu = eps = 1, dmu given as argument, default 1): H = sum_a n_a + x sum_contacts s_ab m_a m_b with
m_a = sigma_x + dmu n_a. The static beta tensor follows from the sum over all eigenstates,
beta_ijk = sum over the six index orders of sum_{p,q>0} M_0p (M - M_00)_pq M_q0 / (E_p E_q), in units of
beta0 = 6 mu^2 dmu / eps^2. rho(x) is evaluated at small +x and -x and kappa is the central difference."""
import itertools
import json
import math
import os
import sys
import numpy as np

L = json.load(open(os.path.join(os.path.dirname(os.path.abspath(__file__)), "layout.json")))
dmu = float(sys.argv[1]) if len(sys.argv) > 1 else 1.0
rot = L["rot_deg"]
ang = [m["u_deg_lattice"] for m in L["molecules"]]
cen = [m["centre_px"] for m in L["molecules"]]
agg = [m["aggregate"] for m in L["molecules"]]
U = [np.array([math.cos(math.radians(a)), math.sin(math.radians(a))]) for a in ang]
sx, nn, I2 = np.array([[0, 1.0], [1, 0]]), np.array([[0, 0], [0, 1.0]]), np.eye(2)


def op(o, a, n):
    m = np.eye(1)
    for k in range(n):
        m = np.kron(m, o if k == a else I2)
    return m


def sign(a, b):
    ua = np.array([math.cos(math.radians(ang[a] + rot)), math.sin(math.radians(ang[a] + rot))])
    ub = np.array([math.cos(math.radians(ang[b] + rot)), math.sin(math.radians(ang[b] + rot))])
    r = np.array([cen[b][0] - cen[a][0], -(cen[b][1] - cen[a][1])])
    r = r / np.hypot(*r)
    return 1 if ua @ ub - 3 * (ua @ r) * (ub @ r) > 0 else -1


def beta(x):
    B = np.zeros((2, 2, 2))
    for g in sorted(set(agg)):
        idx = [i for i in range(len(U)) if agg[i] == g]
        n = len(idx)
        d = [op(sx, k, n) + dmu * op(nn, k, n) for k in range(n)]
        H = sum(op(nn, k, n) for k in range(n))
        for a, b in L["contacts"]:
            if a in idx:
                H = H + sign(a, b) * x * d[idx.index(a)] @ d[idx.index(b)]
        E, C = np.linalg.eigh(H)
        E = E - E[0]
        Me = [C.T @ sum(U[a][i] * d[k] for k, a in enumerate(idx)) @ C for i in range(2)]
        N = len(E)
        T = np.zeros((2, 2, 2))
        for i, j, k in itertools.product(range(2), repeat=3):
            mb = Me[j] - Me[j][0, 0] * np.eye(N)
            T[i, j, k] = sum(Me[i][0, p] * mb[p, q] * Me[k][q, 0] / (E[p] * E[q]) for p in range(1, N) for q in range(1, N))
        B += sum(np.transpose(T, p) for p in itertools.permutations(range(3))) / (6 * dmu)
    return B


def rho(B):
    v = np.einsum("ijj->i", B)
    return math.sqrt(((B ** 2).sum() - 0.6 * v @ v) / (0.6 * v @ v))


h = 1e-4
print("dmu/mu = %.2f: rho(0) = %.5f, kappa = (rho(+h) - rho(-h)) / 2h = %.4f" % (dmu, rho(beta(0.0)), (rho(beta(h)) - rho(beta(-h))) / (2 * h)))
for x in (0.01, 0.02):
    print("x = %.2f: exact rho %.4f" % (x, rho(beta(x))))
----- end of exact.py -----

5. Directions and contacts read back from the PNG
=================================================

Command (run inside verification/): python3 readback.py

----- readback.py -----
"""Read the figure back from image.png alone: lobes as local maxima of the smoothed green + blue signal, the two lobes
of a molecule paired by distance (about 34 px), the brighter lobe taken as the donor end, the acceptor-to-donor
direction binned on the lattice axes, and contacts as pairs of molecules whose nearest lobes are less than 80 px
(2 nm) apart. Compared with layout.json."""
import json
import math
import os
import sys
import numpy as np
from PIL import Image
from scipy import ndimage

here = os.path.dirname(os.path.abspath(__file__))
png = sys.argv[1] if len(sys.argv) > 1 else os.path.join(here, "..", "image.png")
lay = json.load(open(os.path.join(here, "layout.json")))
a = np.asarray(Image.open(png).convert("RGB")).astype(float)
z = ndimage.gaussian_filter(a.sum(axis=2), 3.0)
z[a.shape[0] - 120:, a.shape[1] - 300:] = 0
mx = (z == ndimage.maximum_filter(z, size=15)) & (z > 0.5 * z.max())
ys, xs = np.nonzero(mx)
pts = np.array([xs, ys]).T.astype(float)
val = z[ys, xs]
print("lobes found:", len(pts))
used, mol = set(), []
for i in range(len(pts)):
    if i in used:
        continue
    d = np.hypot(*(pts - pts[i]).T)
    d[i] = 1e9
    j = int(d.argmin())
    used.update((i, j))
    hi, lo = (i, j) if val[i] > val[j] else (j, i)
    v = pts[hi] - pts[lo]
    deg = math.degrees(math.atan2(-v[1], v[0])) - lay["rot_deg"]
    mol.append({"c": pts[[i, j]].mean(axis=0), "lobes": pts[[i, j]], "deg": round(deg / 60) * 60 % 360, "sep": d[j], "ratio": val[lo] / val[hi]})
print("molecules:", len(mol), " lobe separation px: %.1f to %.1f" % (min(m["sep"] for m in mol), max(m["sep"] for m in mol)))
print("dim/bright lobe signal ratio: %.3f to %.3f" % (min(m["ratio"] for m in mol), max(m["ratio"] for m in mol)))
C = np.array([m["centre_px"] for m in lay["molecules"]])
order = [int(np.hypot(*(np.array([m["c"] for m in mol]) - c).T).argmin()) for c in C]
read = [mol[i]["deg"] for i in order]
print("directions that differ from the layout:", sum(r != m["u_deg_lattice"] for r, m in zip(read, lay["molecules"])))
con, gaps, others = [], [], []
for i in range(len(order)):
    for j in range(i + 1, len(order)):
        g = min(np.hypot(*(p - q)) for p in mol[order[i]]["lobes"] for q in mol[order[j]]["lobes"])
        if g < 80:
            con.append([i, j])
            gaps.append(g)
        else:
            others.append(g)
print("contacts read:", len(con), "equal to the layout list:", sorted(con) == sorted(lay["contacts"]))
print("nearest lobes of molecules in contact: %.0f to %.0f px (%.2f to %.2f nm)" % (min(gaps), max(gaps), min(gaps) / 40, max(gaps) / 40))
print("nearest lobes of any other pair: %.0f px (%.2f nm) or more" % (min(others), min(others) / 40))
B = np.array([m["bright_lobe_px"] for m in lay["molecules"]]).round().astype(int)
Dm = np.array([m["dim_lobe_px"] for m in lay["molecules"]]).round().astype(int)
print("donor lobe  green %d to %d, blue %d to %d" % (a[B[:, 1], B[:, 0], 1].min(), a[B[:, 1], B[:, 0], 1].max(), a[B[:, 1], B[:, 0], 2].min(), a[B[:, 1], B[:, 0], 2].max()))
print("acceptor lobe green %d to %d, blue %d to %d" % (a[Dm[:, 1], Dm[:, 0], 1].min(), a[Dm[:, 1], Dm[:, 0], 1].max(), a[Dm[:, 1], Dm[:, 0], 2].min(), a[Dm[:, 1], Dm[:, 0], 2].max()))
----- end of readback.py -----
