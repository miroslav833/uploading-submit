rows = """NYYYNNN.Y.YY
N.NYNNNYYYNN
NNNYYNYNYNNY
N..Y.Y.NYNYY
YNNYYYY.YYNY
YYYYYNYN.NNY
NYNYNNNYNNNN
N.YYNNNYYYNY
NNN.NY.NYNNN
NNN.N.NNN.YN""".split()
R, C = len(rows), len(rows[0])
def nb(r, c):                      # rows 1, 3, 5, ... (index 0, 2, ...) are the unshifted ones
    d = -1 if r % 2 == 0 else 0
    cand = [(r, c - 1), (r, c + 1), (r - 1, c + d), (r - 1, c + d + 1), (r + 1, c + d), (r + 1, c + d + 1)]
    return [(a, b) for a, b in cand if 0 <= a < R and 0 <= b < C and rows[a][b] != "."]
def fill(start, kind):
    seen = {s for s in start if rows[s[0]][s[1]] == kind}
    todo = list(seen)
    while todo:
        for q in nb(*todo.pop()):
            if rows[q[0]][q[1]] == kind and q not in seen:
                seen.add(q); todo.append(q)
    return seen
ysz = fill([(r, 0) for r in range(0, R, 2)], "Y")          # first particle of rows 1, 3, ... on the electrolyte
ni = fill([(r, C - 1) for r in range(1, R, 2)], "N")       # last particle of rows 2, 4, ... on the collector
work = sum(1 for p in ni for q in nb(*p) if q in ysz)
allc = sum(1 for r in range(R) for c in range(C) if rows[r][c] == "N" for q in nb(r, c) if rows[q[0]][q[1]] == "Y")
n_ni = sum(row.count("N") for row in rows)
print(n_ni, sum(row.count("Y") for row in rows), len(ni), len(ysz), allc, work, round(work / n_ni, 3), round(work / (n_ni + 1), 3), round(allc / n_ni, 2))
